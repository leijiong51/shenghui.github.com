let currentLang = localStorage.getItem('sh_lang') || 'zh';
function setLanguage(lang) {
  currentLang = lang;
  localStorage.setItem('sh_lang', lang);
  document.documentElement.lang = lang === 'zh' ? 'zh-CN' : 'lo';
  document.body.setAttribute('lang', lang);
  document.querySelectorAll('[data-i18n]').forEach(function(el) {
    var key = el.getAttribute('data-i18n');
    if (i18nDict[lang][key]) el.innerHTML = i18nDict[lang][key];
  });
  document.querySelectorAll('.lang-btn').forEach(function(b) { b.classList.toggle('active', b.dataset.lang === lang); });
  renderProducts();
}
function navigate(route) { location.hash = route; }
function handleRoute() {
  var hash = location.hash.replace('#', '') || 'home';
  var valid = ['home','products','about','contact','admin'];
  var route = valid.indexOf(hash) > -1 ? hash : 'home';
  document.querySelectorAll('.view').forEach(function(v) { v.classList.remove('active'); });
  var view = document.getElementById('view-' + route);
  if (view) view.classList.add('active');
  document.querySelectorAll('.nav-link[data-route]').forEach(function(l) { l.classList.toggle('active', l.dataset.route === route); });
  var isAdmin = route === 'admin';
  document.getElementById('siteHeader').style.display = isAdmin ? 'none' : '';
  document.getElementById('siteFooter').style.display = isAdmin ? 'none' : '';
  document.getElementById('floatingWhatsapp').style.display = isAdmin ? 'none' : '';
  if (!isAdmin) window.scrollTo(0, 0);
  if (route === 'products') renderProducts();
  if (route === 'admin') checkAdminAuth();
}
window.addEventListener('hashchange', handleRoute);

function submitQuoteForm(e) {
  e.preventDefault();
  var data = Object.fromEntries(new FormData(e.target));
  data.type = 'quote'; data.date = new Date().toISOString(); data.read = false;
  var list = JSON.parse(localStorage.getItem('sh_inquiries') || '[]');
  list.unshift(data); localStorage.setItem('sh_inquiries', JSON.stringify(list));

}

  type = type || 'success';
  var c = document.getElementById('toastContainer');
  var t = document.createElement('div');
  t.className = 'toast' + (type === 'error' ? ' error' : '');
  t.innerHTML = '<svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="'+(type==='error'?'#dc2626':'#22c55e')+'" stroke-width="2"><path d="M22 11.08V12a10 10 0 11-5.93-9.14"/><polyline points="22 4 12 14.01 9 11.01"/></svg><span>'+msg+'</span>';
  c.appendChild(t);
  setTimeout(function() { t.remove(); }, 4000);
}

function updateDashboard() {
  var inquiries = JSON.parse(localStorage.getItem('sh_inquiries') || '[]');
  document.getElementById('statProducts').textContent = products.length;
  document.getElementById('statInquiries').textContent = inquiries.length;
  document.getElementById('statRead').textContent = inquiries.filter(function(i){return i.read;}).length;
  document.getElementById('statUnread').textContent = inquiries.filter(function(i){return !i.read;}).length;
  var tb = document.getElementById('recentInquiriesTable');
  var recent = inquiries.slice(0, 5);
  if (recent.length === 0) { tb.innerHTML = '<tr><td colspan="5" style="text-align:center;padding:32px;color:var(--text-light);">暂无询盘</td></tr>'; }
  else {
    tb.innerHTML = recent.map(function(i) {
      return '<tr><td>'+(i.name||'-')+'</td><td>'+(i.company||'-')+'</td><td>'+(i.product||i.subject||'-')+'</td><td>'+new Date(i.date).toLocaleDateString()+'</td><td><span class="badge '+(i.read?'badge-success':'badge-warning')+'">'+(i.read?'已读':'新')+'</span></td></tr>';
    }).join('');
  }
}
function renderAdminProducts() {
  var tb = document.getElementById('adminProductsTable');
  if (products.length === 0) { tb.innerHTML = '<tr><td colspan="6" style="text-align:center;padding:32px;">暂无产品</td></tr>'; return; }
  tb.innerHTML = products.map(function(p) {
    return '<tr><td><img src="'+p.image+'" class="table-product-img" alt="'+p.name+'"></td><td><strong>'+p.name+'</strong></td><td><span class="badge badge-warning">'+(p.category==='radish'?'萝卜干':'茶叶')+'</span></td><td>'+p.moq+'</td><td><span class="badge '+(p.status==='active'?'badge-success':'badge-danger')+'">'+(p.status==='active'?'上架':'下架')+'</span></td><td><div class="action-btns"><button class="action-btn edit" onclick="editProduct('+p.id+')"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M12 20h9"/><path d="M16.5 3.5a2.121 2.121 0 013 3L7 19l-4 1 1-4L16.5 3.5z"/></svg></button><button class="action-btn delete" onclick="deleteProduct('+p.id+')"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><polyline points="3 6 5 6 21 6"/><path d="M19 6v14a2 2 0 01-2 2H7a2 2 0 01-2-2V6m3 0V4a2 2 0 012-2h4a2 2 0 012 2v2"/></svg></button></div></td></tr>';
  }).join('');
}
function openProductModal(productId) {
  var p = productId ? products.find(function(x){return x.id===productId;}) : null;
  var isEdit = !!p;
  var cats = [{v:'radish',l:'萝卜干制品'},{v:'tea',l:'贵州茶叶'}];
  var catOpts = cats.map(function(c){return '<option value="'+c.v+'"'+(p&&p.category===c.v?' selected':'')+'>'+c.l+'</option>';}).join('');
  var m = document.createElement('div');
  m.className = 'modal-overlay';
  m.innerHTML = '<div class="modal"><div class="modal-header"><h3>'+(isEdit?'编辑产品':'新增产品')+'</h3><button class="modal-close" onclick="closeModal()"><svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg></button></div><div class="modal-body"><div class="form-row"><div class="form-group"><label>产品名称（中文）*</label><input type="text" id="pName" value="'+(p?p.name:'')+'" required></div><div class="form-group"><label>产品名称（老挝语）</label><input type="text" id="pNameLo" value="'+(p?p.nameLo:'')+'"></div></div><div class="form-row"><div class="form-group"><label>产品分类 *</label><select id="pCategory" required>'+catOpts+'</select></div><div class="form-group"><label>起订量</label><input type="text" id="pMoq" value="'+(p?p.moq:'500kg')+'"></div></div><div class="form-group"><label>图片URL *</label><input type="text" id="pImage" value="'+(p?p.image:'')+'" required></div><div class="form-row"><div class="form-group"><label>产地</label><input type="text" id="pOrigin" value="'+(p?p.origin:'贵州, 中国')+'"></div><div class="form-group"><label>状态</label><select id="pStatus"><option value="active"'+(p&&p.status!=='inactive'?' selected':'')+'>上架</option><option value="inactive"'+(p&&p.status==='inactive'?' selected':'')+'>下架</option></select></div></div><div class="form-row"><div class="form-group"><label>规格</label><input type="text" id="pSpec" value="'+(p?p.spec:'')+'"></div><div class="form-group"><label>认证</label><input type="text" id="pCert" value="'+(p?p.cert:'HACCP, ISO22000')+'"></div></div></div><div class="modal-footer"><button class="btn btn-sm" style="background:var(--bg-gray);color:var(--text);" onclick="closeModal()">取消</button><button class="btn btn-primary btn-sm" onclick="saveProduct('+(productId||'null')+')">'+(isEdit?'保存修改':'创建产品')+'</button></div></div>';
  document.getElementById('modalContainer').appendChild(m);
}
function editProduct(id) { openProductModal(id); }
function saveProduct(id) {
  var data = {name:document.getElementById('pName').value,nameLo:document.getElementById('pNameLo').value,category:document.getElementById('pCategory').value,image:document.getElementById('pImage').value,moq:document.getElementById('pMoq').value,origin:document.getElementById('pOrigin').value,spec:document.getElementById('pSpec').value,cert:document.getElementById('pCert').value,status:document.getElementById('pStatus').value};
  if (!data.name || !data.image) { showToast('产品名称和图片URL为必填项', 'error'); return; }
  if (id) { var idx = products.findIndex(function(p){return p.id===id;}); if (idx>-1) products[idx] = Object.assign({}, products[idx], data); showToast('产品已更新'); }
  else { data.id = Date.now(); products.push(data); showToast('产品已创建'); }
  saveProducts(); closeModal(); renderAdminProducts(); updateDashboard();
}
function deleteProduct(id) {
  if (!confirm('确定要删除该产品吗？')) return;
  products = products.filter(function(p){return p.id!==id;});
  saveProducts(); renderAdminProducts(); updateDashboard(); showToast('产品已删除');
}
function renderAdminInquiries() {
  var tb = document.getElementById('adminInquiriesTable');
  var inquiries = JSON.parse(localStorage.getItem('sh_inquiries') || '[]');
  if (inquiries.length === 0) { tb.innerHTML = '<tr><td colspan="7" style="text-align:center;padding:32px;">暂无询盘</td></tr>'; return; }
  tb.innerHTML = inquiries.map(function(i, idx) {
    return '<tr><td><strong>'+(i.name||'-')+'</strong></td><td>'+(i.company||'-')+'</td><td>'+(i.email||'-')+'</td><td>'+(i.product||i.subject||'-')+'</td><td>'+new Date(i.date).toLocaleDateString()+'</td><td><span class="badge '+(i.read?'badge-success':'badge-warning')+'">'+(i.read?'已读':'未读')+'</span></td><td><div class="action-btns"><button class="action-btn edit" onclick="viewInquiry('+idx+')"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M1 12s4-8 11-8 11 8 11 8-4 8-11 8-11-8-11-8z"/><circle cx="12" cy="12" r="3"/></svg></button><button class="action-btn delete" onclick="deleteInquiry('+idx+')"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><polyline points="3 6 5 6 21 6"/><path d="M19 6v14a2 2 0 01-2 2H7a2 2 0 01-2-2V6m3 0V4a2 2 0 012-2h4a2 2 0 012 2v2"/></svg></button></div></td></tr>';
  }).join('');
}
function viewInquiry(idx) {
  var inquiries = JSON.parse(localStorage.getItem('sh_inquiries') || '[]');
  var i = inquiries[idx];
  if (!i.read) { i.read = true; localStorage.setItem('sh_inquiries', JSON.stringify(inquiries)); }
  var m = document.createElement('div');
  m.className = 'modal-overlay';
  m.innerHTML = '<div class="modal"><div class="modal-header"><h3>询盘详情</h3><button class="modal-close" onclick="closeModal()"><svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg></button></div><div class="modal-body"><div style="display:grid;grid-template-columns:1fr 1fr;gap:16px;margin-bottom:16px;"><div><strong>姓名：</strong>'+(i.name||'-')+'</div><div><strong>公司：</strong>'+(i.company||'-')+'</div><div><strong>邮箱：</strong>'+(i.email||'-')+'</div><div><strong>电话：</strong>'+(i.phone||'-')+'</div><div><strong>产品：</strong>'+(i.product||'-')+'</div><div><strong>日期：</strong>'+new Date(i.date).toLocaleString()+'</div></div>'+(i.subject?'<div style="margin-bottom:12px;"><strong>主题：</strong>'+i.subject+'</div>':'')+'<div><strong>详细需求：</strong><p style="margin-top:8px;padding:12px;background:var(--bg-gray);border-radius:8px;white-space:pre-wrap;">'+(i.message||'-')+'</p></div></div><div class="modal-footer"><a href="mailto:'+i.email+'" class="btn btn-primary btn-sm">邮件回复</a><button class="btn btn-sm" style="background:var(--bg-gray);color:var(--text);" onclick="closeModal()">关闭</button></div></div>';
  document.getElementById('modalContainer').appendChild(m);
  renderAdminInquiries(); updateDashboard();
}
function deleteInquiry(idx) {
  if (!confirm('确定要删除该询盘吗？')) return;
  var inquiries = JSON.parse(localStorage.getItem('sh_inquiries') || '[]');
  inquiries.splice(idx, 1); localStorage.setItem('sh_inquiries', JSON.stringify(inquiries));
  renderAdminInquiries(); updateDashboard(); showToast('询盘已删除');
}
function closeModal() { var m = document.querySelector('.modal-overlay'); if (m) m.remove(); }

  showToast('公司信息已保存');
}
